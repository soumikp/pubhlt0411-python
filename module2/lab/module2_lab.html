A default is a claim about the world

Returning 0 for an absent month asserts that nobody died that month. That is correct when the month is genuinely outside the reporting period, and wrong when the month is missing because no one reported. Module 4 treats missing data as its own category.

def summarize(county, deaths, population):
    """Return the county's rate and a classification against 250 per 100,000."""
    rate = deaths / population * 100000
    if rate > 250:
        classification = "High"
    else:
        classification = "Moderate"
    return rate, classification

Your turn. Complete the method so it returns the county’s rate.

# The blank is self — inside a method, that is how the object refers to its own data.
# McKean's rate is 493.4 per 100,000.
# Note the parentheses on rate_per_100k() — without them you get the function, not the number.
class County: """A Pennsylvania county with COVID-19 mortality data.""" def __init__(self, name, deaths, population): self.name = name self.deaths = deaths self.population = population def rate_per_100k(self): """Return deaths per 100,000 residents.""" return self.deaths / self.population * 100000 mckean = County("McKean", 195, 39519) print(f"{mckean.name}: {mckean.rate_per_100k():.1f} per 100,000") 
class County:
    """A Pennsylvania county with COVID-19 mortality data."""

    def __init__(self, name, deaths, population):
        self.name = name
        self.deaths = deaths
        self.population = population

    def rate_per_100k(self):
        """Return deaths per 100,000 residents."""
        return self.deaths / self.population * 100000


mckean = County("McKean", 195, 39519)

print(f"{mckean.name}: {mckean.rate_per_100k():.1f} per 100,000")
